318:def get_auto_color_masks(img, roi_height_ratio=0.35, min_bright=180, color_quant=16, top_colors=3):
319:    """
320:    Автоматически находит самые яркие и часто встречающиеся цвета в нижней части изображения
321:    (без sklearn, только numpy!). Маски по цвету + яркость.
322:    color_quant — степень округления цветов (16: цвета округляются к ближайшему кратному 16)
323:    """
324:    img_np = np.array(img.convert('RGB'))
325:    h, w, _ = img_np.shape
326:    roi = img_np[int(h*(1-roi_height_ratio)):, :, :]
328:    # Маска по яркости (белый, светлый логотип)
329:    gray = cv2.cvtColor(roi, cv2.COLOR_RGB2GRAY)

332:    # Грубое квантование цвета (убираем шум)
333:    roi_quant = (roi // color_quant) * color_quant
334:    roi_flat = roi_quant.reshape(-1, 3)
335:    # Получаем уникальные цвета и их частоты
336:    uniq, counts = np.unique(roi_flat, axis=0, return_counts=True)
337:    idxs = np.argsort(-counts)[:top_colors]
338:    dominant_colors = uniq[idxs]
340:    # Строим маски для каждого доминирующего цвета
341:    mask_colors = np.zeros_like(mask_bright)
342:    for color in dominant_colors:
343:        lower = np.clip(color - color_quant, 0, 255)
344:        upper = np.clip(color + color_quant, 0, 255)
345:        mask = cv2.inRange(roi, lower, upper)

348:    # Маска логотипа: яркие области + цветовые
349:    mask_logo = cv2.bitwise_or(mask_bright, mask_colors)
350:    # Маску возвращаем в размер всего изображения
351:    full_mask = np.zeros((h, w), dtype=np.uint8)
352:    full_mask[int(h*(1-roi_height_ratio)):, :] = mask_logo
353:    return full_mask
