322:def get_auto_color_masks(img, roi_height_ratio=0.35, min_bright=180, color_quant=16, top_colors=3):
323:    """
324:    Автоматически находит самые яркие и часто встречающиеся цвета в нижней части изображения
325:    (без sklearn, только numpy!). Маски по цвету + яркость.
326:    color_quant — степень округления цветов (16: цвета округляются к ближайшему кратному 16)
327:    """
328:    img_np = np.array(img.convert('RGB'))
329:    h, w, _ = img_np.shape
330:    roi = img_np[int(h*(1-roi_height_ratio)):, :, :]
332:    # Маска по яркости (белый, светлый логотип)
333:    gray = cv2.cvtColor(roi, cv2.COLOR_RGB2GRAY)
334:    _, mask_bright = cv2.threshold(gray, min_bright, 255, cv2.THRESH_BINARY)
336:    # Грубое квантование цвета (убираем шум)
337:    roi_quant = (roi // color_quant) * color_quant
338:    roi_flat = roi_quant.reshape(-1, 3)
339:    # Получаем уникальные цвета и их частоты
340:    uniq, counts = np.unique(roi_flat, axis=0, return_counts=True)
341:    idxs = np.argsort(-counts)[:top_colors]
342:    dominant_colors = uniq[idxs]
344:    # Строим маски для каждого доминирующего цвета
345:    mask_colors = np.zeros_like(mask_bright)
346:    for color in dominant_colors:
347:        lower = np.clip(color - color_quant, 0, 255)
348:        upper = np.clip(color + color_quant, 0, 255)
349:        mask = cv2.inRange(roi, lower, upper)
350:        mask_colors = cv2.bitwise_or(mask_colors, mask)
352:    # Маска логотипа: яркие области + цветовые
353:    mask_logo = cv2.bitwise_or(mask_bright, mask_colors)
354:    # Маску возвращаем в размер всего изображения
355:    full_mask = np.zeros((h, w), dtype=np.uint8)
356:    full_mask[int(h*(1-roi_height_ratio)):, :] = mask_logo
357:    return full_mask
